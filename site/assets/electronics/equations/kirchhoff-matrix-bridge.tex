\[
\begin{gathered}
\text{Equation 1: }5I_1-2I_2=10\\[0.22cm]
\text{Equation 2: }-2I_1+7I_2=6
\end{gathered}
\quad\Longrightarrow\quad
\begin{gathered}
\begin{array}{c|cc|c}
 & I_1 & I_2 & \text{right side}\\ \hline
\text{row 1} & 5 & -2 & 10\\
\text{row 2} & -2 & 7 & 6
\end{array}
\end{gathered}
\]
\[
\underbrace{\begin{bmatrix}5&-2\\-2&7\end{bmatrix}}_{A\,[\mathrm{k\Omega}]}
\underbrace{\begin{bmatrix}I_1\\I_2\end{bmatrix}}_{\mathbf{x}\,[\mathrm{mA}]}
=
\underbrace{\begin{bmatrix}10\\6\end{bmatrix}}_{\mathbf{b}\,[\mathrm{V}]}
\quad\Longleftrightarrow\quad
\begin{bmatrix}5I_1-2I_2\\-2I_1+7I_2\end{bmatrix}
=\begin{bmatrix}10\\6\end{bmatrix}
\]
