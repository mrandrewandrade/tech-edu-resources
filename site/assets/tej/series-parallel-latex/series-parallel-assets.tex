\equationpage
  {\mixedcircuit{10.0\unit{V}}{1.0\unit{k\Omega}}{2.0\unit{k\Omega}}{2.0\unit{k\Omega}}{}}
  {\dfrac{1}{R_P}&=\dfrac{1}{R_a}+\dfrac{1}{R_b}}
  {R_T&=R_S+R_P \qquad I_T=\dfrac{V_T}{R_T}}
  {V&=IR \qquad I_T=I_a+I_b}
  {R_{23}&=\dfrac{R_2R_3}{R_2+R_3} \qquad R_T=R_1+R_{23}}
  {I_T&=\dfrac{V_T}{R_T} \qquad V_{23}=V_T-I_TR_1 \qquad I_T=I_2+I_3}

% Page 2: guided example substitution, solution, and checks
\thispagestyle{empty}
\solvestep{1 | EQUIVALENT RESISTANCE}
{\Large
\begin{align*}
R_{23}&=\frac{R_2R_3}{R_2+R_3}
=\frac{(2.0\unit{k\Omega})(2.0\unit{k\Omega})}{2.0\unit{k\Omega}+2.0\unit{k\Omega}}
=1.0\unit{k\Omega}\\[0.06in]
R_T&=R_1+R_{23}=1.0\unit{k\Omega}+1.0\unit{k\Omega}=2.0\unit{k\Omega}
\end{align*}}
\begin{minipage}[t]{0.48\textwidth}
\solvestep{2 | CURRENT AND VOLTAGES}
{\Large
\begin{align*}
I_T&=\frac{V_T}{R_T}=\frac{10.0\unit{V}}{2.0\unit{k\Omega}}=5.0\unit{mA}\\[0.05in]
V_1&=I_TR_1=(5.0\unit{mA})(1.0\unit{k\Omega})=5.0\unit{V}\\[0.05in]
V_{23}&=V_T-V_1=10.0\unit{V}-5.0\unit{V}=5.0\unit{V}
\end{align*}}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\solvestep{3 | BRANCH CURRENTS AND CHECKS}
{\Large
\begin{align*}
I_2=I_3&=\frac{V_{23}}{R_2}=\frac{5.0\unit{V}}{2.0\unit{k\Omega}}=2.5\unit{mA}
\end{align*}}
{\large\bfseries
\begin{align*}
V_1+V_{23}&=5.0\unit{V}+5.0\unit{V}=10.0\unit{V}\quad\checkmark\\
I_2+I_3&=2.5\unit{mA}+2.5\unit{mA}=5.0\unit{mA}\quad\checkmark
\end{align*}}
\end{minipage}\par
\vspace{0.03in}
\solvestep{FINAL ANSWERS}
\begin{center}
\setlength{\tabcolsep}{0.15in}
\renewcommand{\arraystretch}{1.28}
{\LARGE\begin{tabular}{|c|c|c|c|c|c|}
\hline
$R_{23}$ & $\finalanswer{1.0\unit{k\Omega}}$ & $R_T$ & $\finalanswer{2.0\unit{k\Omega}}$ & $I_T$ & $\finalanswer{5.0\unit{mA}}$ \\
\hline
$V_1$ & $\finalanswer{5.0\unit{V}}$ & $V_{23}$ & $\finalanswer{5.0\unit{V}}$ & $I_2=I_3$ & $\finalanswer{2.5\unit{mA}}$ \\
\hline
\end{tabular}}
\end{center}

\equationpage
  {\mixedcircuit{18.0\unit{V}}{3.0\unit{k\Omega}}{6.0\unit{k\Omega}}{3.0\unit{k\Omega}}{}}
  {\dfrac{1}{R_P}&=\dfrac{1}{R_a}+\dfrac{1}{R_b}}
  {R_T&=R_S+R_P \qquad I_T=\dfrac{V_T}{R_T}}
  {V&=IR \qquad I_T=I_a+I_b \qquad P_T=V_TI_T}
  {R_{23}&=\dfrac{R_2R_3}{R_2+R_3} \qquad R_T=R_1+R_{23}}
  {I_T&=\dfrac{V_T}{R_T} \qquad V_{23}=V_T-I_TR_1 \qquad P_T=V_TI_T}

% Page 4: Question 2 substitution, solution, and checks
\thispagestyle{empty}
\solvestep{1 | EQUIVALENT RESISTANCE}
{\LARGE
\begin{align*}
R_{23}&=\frac{R_2R_3}{R_2+R_3}
=\frac{(6.0\unit{k\Omega})(3.0\unit{k\Omega})}{6.0\unit{k\Omega}+3.0\unit{k\Omega}}
=2.0\unit{k\Omega}\\[0.06in]
R_T&=R_1+R_{23}=3.0\unit{k\Omega}+2.0\unit{k\Omega}=5.0\unit{k\Omega}
\end{align*}}
\begin{minipage}[t]{0.48\textwidth}
\solvestep{2 | TOTAL CURRENT AND POWER}
{\Large
\begin{align*}
I_T&=\frac{V_T}{R_T}=\frac{18.0\unit{V}}{5.0\unit{k\Omega}}=3.60\unit{mA}\\[0.05in]
P_T&=V_TI_T=(18.0\unit{V})(3.60\unit{mA})=64.8\unit{mW}
\end{align*}}
\solvestep{3 | SECTION VOLTAGES}
{\Large
\begin{align*}
V_1&=I_TR_1=(3.60\unit{mA})(3.0\unit{k\Omega})=10.8\unit{V}\\[0.05in]
V_{23}&=V_T-V_1=18.0\unit{V}-10.8\unit{V}=7.2\unit{V}
\end{align*}}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\solvestep{4 | BRANCH CURRENTS}
{\Large
\begin{align*}
I_2&=\frac{V_{23}}{R_2}=\frac{7.2\unit{V}}{6.0\unit{k\Omega}}=1.20\unit{mA}\\[0.05in]
I_3&=\frac{V_{23}}{R_3}=\frac{7.2\unit{V}}{3.0\unit{k\Omega}}=2.40\unit{mA}
\end{align*}}
\end{minipage}\par

% Page 5: Question 4 equations
\equationpage
  {\mixedcircuit{20.0\unit{V}}{1.2\unit{k\Omega}}{2.4\unit{k\Omega}}{?}{I_T=10.0\unit{mA}}}
  {V_S&=I_TR_S \qquad V_P=V_T-V_S}
  {I_a&=\dfrac{V_P}{R_a} \qquad I_b=I_T-I_a}
  {R_b&=\dfrac{V_P}{I_b} \qquad R_T=\dfrac{V_T}{I_T}}
  {V_1&=I_TR_1 \qquad V_{23}=V_T-V_1 \qquad I_2=\dfrac{V_{23}}{R_2}}
  {I_3&=I_T-I_2 \qquad R_3=\dfrac{V_{23}}{I_3} \qquad R_T=\dfrac{V_T}{I_T}}

% Page 6: Question 4 substitution, solution, and checks
\thispagestyle{empty}
\begin{minipage}[t]{0.48\textwidth}
\solvestep{1 | SECTION VOLTAGES}
{\LARGE
\begin{align*}
V_1&=I_TR_1=(10.0\unit{mA})(1.2\unit{k\Omega})=12.0\unit{V}\\[0.07in]
V_{23}&=V_T-V_1=20.0\unit{V}-12.0\unit{V}=8.0\unit{V}
\end{align*}}
\solvestep{2 | BRANCH CURRENTS}
{\LARGE
\begin{align*}
I_2&=\frac{V_{23}}{R_2}=\frac{8.0\unit{V}}{2.4\unit{k\Omega}}=3.33\unit{mA}\\[0.07in]
I_3&=I_T-I_2=10.0\unit{mA}-3.33\unit{mA}=6.67\unit{mA}
\end{align*}}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\solvestep{3 | UNKNOWN RESISTOR}
{\LARGE
\begin{align*}
R_3&=\frac{V_{23}}{I_3}=\frac{8.0\unit{V}}{6.67\unit{mA}}\approx1.20\unit{k\Omega}
\end{align*}}
\solvestep{4 | TOTAL RESISTANCE}
{\LARGE
\begin{align*}
R_T&=\frac{V_T}{I_T}=\frac{20.0\unit{V}}{10.0\unit{mA}}=2.00\unit{k\Omega}
\end{align*}}
\solvestep{CHECK}
{\large\bfseries
\begin{align*}
I_2+I_3&=3.33\unit{mA}+6.67\unit{mA}=10.0\unit{mA}\quad\checkmark
\end{align*}}
\end{minipage}\par
