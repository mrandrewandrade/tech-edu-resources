\equationpage
  {\mixedcircuit{10.0\unit{V}}{1.0\unit{k\Omega}}{2.0\unit{k\Omega}}{2.0\unit{k\Omega}}{}}
  {\dfrac{1}{R_P}&=\dfrac{1}{R_a}+\dfrac{1}{R_b}}
  {R_T&=R_S+R_P \qquad I_T=\dfrac{V_T}{R_T}}
  {V&=IR \qquad I_T=I_a+I_b}
  {R_{23}&=\dfrac{R_2R_3}{R_2+R_3} \qquad R_T=R_1+R_{23}}
  {I_T&=\dfrac{V_T}{R_T} \qquad V_{23}=V_T-I_TR_1 \qquad I_T=I_2+I_3}

% Page 2: guided example substitution, solution, and checks
\thispagestyle{empty}
\sectionlabel{SUBSTITUTE AND SOLVE}
{\Large
\begin{align*}
R_{23}&=\frac{(2.0\unit{k\Omega})(2.0\unit{k\Omega})}{2.0\unit{k\Omega}+2.0\unit{k\Omega}}=1.0\unit{k\Omega}\\[0.10in]
R_T&=R_1+R_{23}=1.0\unit{k\Omega}+1.0\unit{k\Omega}=2.0\unit{k\Omega}\\[0.10in]
I_T&=\frac{V_T}{R_T}=\frac{10.0\unit{V}}{2.0\unit{k\Omega}}=5.0\unit{mA}\\[0.10in]
V_1&=I_TR_1=(5.0\unit{mA})(1.0\unit{k\Omega})=5.0\unit{V}\\[0.10in]
V_{23}&=V_T-V_1=10.0\unit{V}-5.0\unit{V}=5.0\unit{V}\\[0.10in]
I_2=I_3&=\frac{V_{23}}{2.0\unit{k\Omega}}=2.5\unit{mA}
\end{align*}}
\vspace{0.05in}
{\color{Green}\Large\bfseries
\begin{align*}
V_1+V_{23}&=5.0\unit{V}+5.0\unit{V}=10.0\unit{V}=V_T\quad\checkmark\\
I_2+I_3&=2.5\unit{mA}+2.5\unit{mA}=5.0\unit{mA}=I_T\quad\checkmark
\end{align*}}

{\Large

I_3&=10.0\unit{mA}-3.33\unit{mA}=6.67\unit{mA}\\[0.09in]
R_3&=\frac{8.0\unit{V}}{6.67\unit{mA}}\approx1.20\unit{k\Omega}\\[0.09in]
R_T&=\frac{20.0\unit{V}}{10.0\unit{mA}}=2.00\unit{k\Omega}
\end{align*}}
{\color{Green}\Large\bfseries
\begin{align*}
3.33\unit{mA}+6.67\unit{mA}=10.0\unit{mA}=I_T\quad\checkmark
\end{align*}}
